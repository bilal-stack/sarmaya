# Development-only tooling. Deliberately separate from requirements.txt, which
# the Dockerfile installs — a linter has no business in a production image.
#
#   pip install -r requirements-dev.txt
#   ruff check .            # what would fail
#   ruff check . --fix      # fix the safe ones
#
# The rule set is narrow on purpose; see [tool.ruff] in pyproject.toml for what
# is enabled, what is ignored, and why.
-r requirements.txt

ruff>=0.6,<1

# Regenerates requirements.lock. MUST be run on Linux, which is what the image
# and CI actually use:
#
#   docker run --rm -v "$PWD:/w" -w /w python:3.14-slim sh -c \
#     "pip install -q pip-tools && pip-compile --strip-extras --no-header \
#      --output-file=requirements.lock requirements.txt"
#
# Not on your own machine unless that machine is Linux. pip-compile resolves
# for the platform it runs on, and the difference is not cosmetic: a lock built
# on Windows omits uvloop, which uvicorn uses on Linux, and adds colorama,
# which only exists there. The image would come up either way and quietly run
# on the asyncio event loop instead. CI regenerates and diffs, so a lock built
# on the wrong platform fails the run rather than shipping.
pip-tools>=7.4,<8
