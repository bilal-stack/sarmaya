# Development-only tooling. Deliberately separate from requirements.txt, which
# the Dockerfile installs — a linter has no business in a production image.
#
#   pip install -r requirements-dev.txt
#   ruff check .            # what would fail
#   ruff check . --fix      # fix the safe ones
#
# The rule set is narrow on purpose; see [tool.ruff] in pyproject.toml for what
# is enabled, what is ignored, and why.
-r requirements.txt

ruff>=0.6,<1

# Regenerates requirements.lock from requirements.txt:
#
#   pip-compile --strip-extras --output-file=requirements.lock requirements.txt
#
# The lock is what the Dockerfile installs, so the image is built from exact
# versions rather than from whatever the ranges resolved to that morning. CI
# re-runs the command and fails if the result differs, which is what stops the
# two files drifting apart silently.
pip-tools>=7.4,<8
